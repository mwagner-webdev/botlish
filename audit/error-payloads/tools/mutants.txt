@@ nested-destructuring-handler-specific a handler's nested pattern reads the inner fields from the payload itself (handler-specific rules)
## surface/lower.tcl
<<<<
            set body [concat [Projections $pattern $temp] $body]
====
            set flat {}
            foreach field [dict get $pattern fields] {
                if {[dict get $field nested] ne ""} {
                    foreach inner [dict get $field nested fields] {
                        if {[dict get $inner nested] eq ""} {
                            dict set inner id [dict get $field id]/x
                            lappend flat $inner
                        }
                    }
                } else {
                    lappend flat $field
                }
            }
            set body [concat [Projections [dict replace $pattern fields $flat] $temp] $body]
>>>>
