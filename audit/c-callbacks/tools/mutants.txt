# --- raw argument and result bit preservation --------------------------------

@@ raw-float-argument-promoted a raw C float argument is widened to a double (fpromote) before its bits are stored: the bits a callback sees are a binary64's, a signaling NaN quieted
## native/src/codegen/callback.rs
<<<<
                CAbi::F32 => {
                    let bits = b.ins().bitcast(I32, MemFlagsData::new(), v);
                    b.ins().uextend(I64, bits)
                }
====
                CAbi::F32 => {
                    let wide = b.ins().fpromote(F64, v);
                    b.ins().bitcast(I64, MemFlagsData::new(), wide)
                }
>>>>

@@ raw-result-through-double a raw C float result goes through a double (its bits read as a binary64, then narrowed): NaN payloads and signaling NaNs are not returned as they were
## native/src/codegen/callback.rs
<<<<
                CAbi::F32 => {
                    let bits = b.ins().ireduce(I32, word);
                    b.ins().bitcast(F32, MemFlagsData::new(), bits)
                }
====
                CAbi::F32 => {
                    let wide = b.ins().bitcast(F64, MemFlagsData::new(), word);
                    b.ins().fdemote(F32, wide)
                }
>>>>

@@ raw-nan-canonicalized the runtime canonicalizes a raw double argument's NaN (every NaN becomes the default quiet NaN) before the callback sees it
## native/src/runtime/callback.rs
<<<<
        CAbi::RawF64 => vm.new_raw_float(CFloatFormat::Binary64, word),
====
        CAbi::RawF64 => vm.new_raw_float(CFloatFormat::Binary64, if f64::from_bits(word).is_nan() { 0x7ff8000000000000 } else { word }),
