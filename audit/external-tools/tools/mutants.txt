# Mutants of explicit external tools, target support packages and the
# artifact manifest (audit/external-tools/tools/mutate.tcl;
# EXTERNAL-COMPILER-TOOLS-AND-TARGET-PACKAGES-MILESTONE-1.md, "Mutation
# testing"): each a deliberate bug in what keeps an extension package
# ordinary until it is explicitly run -- activation, resolution, the process
# boundary, dependency correctness, artifact integrity and target
# compatibility -- or in what the sample tools trust.
#
#   @@ NAME DESCRIPTION
#   !! TEST ...        the tests of tests/external-tools.test meant to
#                      detect it (one failing is a kill)
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# A mutant may have several edits. Lines starting with # outside an edit
# are comments. No mutant edits Rust: the copy's native/target is a link to
# this checkout's.

@@ run-on-install an installed tool's program runs when the package is installed (a post-install hook)
!! et-install-runs-nothing
## pm/install.tcl
<<<<
    Commit $ctx $operations $newStateBytes $descriptorBytes $graphBytes
    return [concat $notes [Report $plan $state $operations] $viewLines]
====
    Commit $ctx $operations $newStateBytes $descriptorBytes $graphBytes
    foreach name [dict keys [pm::state::read [dict get $ctx meta]]] {
        set dir [file join [dict get $ctx modules] $name]
        if {[catch {pm::descriptor::read [file join $dir botlish.json] artifact} installed]} continue
        dict for {tool spec} [dict getdef $installed tools {}] {
            set program [file join [file tempdir] p]
            file copy [file join $dir [dict get $spec programs linux-x86_64]] $program
            file attributes $program -permissions 0755
            catch {exec $program install}
        }
    }
    return [concat $notes [Report $plan $state $operations] $viewLines]
