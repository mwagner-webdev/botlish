# Mutants of the golden test framework (audit/golden/tools/mutate.tcl;
# GOLDEN-TEST-FRAMEWORK.md, "Framework mutation results"): each a deliberate
# bug in stdlib/test.bot -- the brief's item 54 list, plus a few the framework's
# own design suggests. Format (as audit/static-compile/tools/mutants.txt):
#
#   @@ NAME DESCRIPTION
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# Every mutant must be killed by tests/golden-framework.test (and the golden
# suites, and the framework fuzzer, which mutate.tcl also runs).

@@ registration-executes registering a case calls its check at once, before storing it
## stdlib/test.bot
<<<<
    suite.cases.push(TestCase {name: name, run: run})
====
    early = run()
    suite.cases.push(TestCase {name: name, run: run})
>>>>

@@ registration-drops-one registering the second case of a suite stores nothing
## stdlib/test.bot
<<<<
    suite.cases.push(TestCase {name: name, run: run})
====
    if std::mutable_vector::length(suite.cases) != 1:
        suite.cases.push(TestCase {name: name, run: run})
>>>>

@@ registration-reorders registering a case puts it first, moving the former first case to the end
## stdlib/test.bot
<<<<
    suite.cases.push(TestCase {name: name, run: run})
====
    entry = TestCase {name: name, run: run}
    if std::mutable_vector::length(suite.cases) == 0:
        suite.cases.push(entry)
    else:
        displaced = suite.cases.swap(0, entry):
            on IndexNotFound:
                entry
        suite.cases.push(displaced)
>>>>

@@ runner-skips-last the runner does not run the last registered case
## stdlib/test.bot
<<<<
    loop c in registered():
        m = (c.run)()
====
    all = registered()
    n = std::list::length(all)
    loop c in all and k from 0 to n:
        if k == n - 1:
            break
        m = (c.run)()
>>>>

@@ suite-starts-with-stale-case a new suite already holds a case, as if one leaked from an earlier suite (the library cannot keep a mutable value outside the context: a module binding is immutable)
## stdlib/test.bot
<<<<

====
    fn stale() -> Match:
        Match {matched: true, expected: "", actual: "", detail: ""}
    Suite {cases: std::mutable_vector::from_list([TestCase {name: "stale", run: stale}])}
>>>>
