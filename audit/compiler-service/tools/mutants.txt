# mutants.txt -- the compiler service's mutants (tools/mutate.tcl).
#
# Format: "@@ NAME DESCRIPTION" starts a mutant; "## FILE" names the file
# its next edits change; each edit is "<<<<", the exact text to replace
# (occurring once), "====", its replacement, ">>>>". Update a mutant's text
# when the code it mutates changes, keeping it a mutant of the same rule.

@@ revision-reads-workspace a revision's source text is the workspace's current document, not the text it analyzed
## service/service.tcl
<<<<
    return [dict get $revision sources $source text]
}
====
    set workspace [Workspace [dict get $revision workspace]]
    set normal [file normalize $source]
    if {[dict exists $workspace overlay $normal text]} {
        return [surface::source::normalize [dict get $workspace overlay $normal text]]
    }
    return [dict get $revision sources $source text]
}
