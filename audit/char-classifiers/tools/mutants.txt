# The Tcl runtime and the registration (core/tclcompat.tcl)

@@ tcl-alpha-is-alnum the Tcl backends classify std::char::is_tcl_alpha as alnum
## core/tclcompat.tcl
<<<<
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alpha [format %c [core::value::charOf $c]]]
====
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alnum [format %c [core::value::charOf $c]]]
>>>>

@@ tcl-bmp-only the Tcl backends' alpha classifies an astral scalar by its low 16 bits
## core/tclcompat.tcl
<<<<
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alpha [format %c [core::value::charOf $c]]]
====
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alpha [format %c [expr {[core::value::charOf $c] & 0xFFFF}]]]
>>>>

@@ tcl-no-type-check the Tcl backends classify a one-character String given to the char form
## core/tclcompat.tcl
<<<<
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alpha [format %c [core::value::charOf $c]]]
====
    if {[core::value::kind $c] eq "str"} { return [isTclAlpha $c] }
    set c [core::value::expect UnicodeChar $c std::char::is_tcl_alpha]
    return [Classify alpha [format %c [core::value::charOf $c]]]
>>>>

# Native lowering and NIR (native/lower.tcl, native/src/nir.rs)

@@ lower-op-swap native lowering gives std::char::is_tcl_alpha the alnum op
## native/lower.tcl
<<<<
        std::char::is_tcl_alpha {op chartclalpha} \
====
        std::char::is_tcl_alpha {op chartclalnum} \
>>>>

@@ lower-str-op native lowering gives std::char::is_tcl_alnum the String form's op
## native/lower.tcl
<<<<
        std::char::is_tcl_alnum {op chartclalnum} \
====
        std::char::is_tcl_alnum {op strtclalnum} \
>>>>

@@ nir-spelling-swap NIR's spellings of the two ops are exchanged (each op reads as the other)
## native/src/nir.rs
<<<<
    "chartclalpha" => CharIsTclAlpha,
    "chartclalnum" => CharIsTclAlnum,
====
    "chartclalpha" => CharIsTclAlnum,
    "chartclalnum" => CharIsTclAlpha,
>>>>

# The inline ASCII rule (native/src/codegen/clif.rs)

@@ clif-no-fold the inline rule does not fold case: only a-z are letters
## native/src/codegen/clif.rs
<<<<
                let folded = self.b.ins().bor_imm_s(cp, 0x20);
====
                let folded = cp;
>>>>

@@ clif-latin1-inline the inline rule takes every Latin-1 scalar (é is no letter inline)
## native/src/codegen/clif.rs
<<<<
                let ascii = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, cp, 0x80);
====
                let ascii = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, cp, 0x100);
>>>>

@@ clif-letter-bound-27 the inline rule counts one scalar past z (and past Z, folded) as a letter
## native/src/codegen/clif.rs
<<<<
                let mut flag = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, offset, 26);
====
                let mut flag = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, offset, 27);
>>>>

@@ clif-digit-signed the inline digit test compares signed: every scalar below 0 is a digit
## native/src/codegen/clif.rs
<<<<
                    let digit = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, digit_offset, 10);
====
                    let digit = self.b.ins().icmp_imm_s(IntCC::SignedLessThan, digit_offset, 10);
>>>>

@@ clif-no-digits the inline alnum rule forgets the digits
## native/src/codegen/clif.rs
<<<<
                    flag = self.b.ins().bor(flag, digit);
====
                    let _ = digit;
>>>>

@@ clif-helper-swap the wide path of alpha calls the alnum helper
## native/src/codegen/clif.rs
<<<<
                let helper = if op == CharIsTclAlpha { "rt_char_is_tcl_alpha" } else { "rt_char_is_tcl_alnum" };
====
                let helper = "rt_char_is_tcl_alnum";
>>>>

@@ clif-word-not-shifted the inline rule reads the tagged word, not the scalar
## native/src/codegen/clif.rs
<<<<
                let cp = self.b.ins().ushr_imm_s(a[0], 3);
                let ascii = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, cp, 0x80);
====
                let cp = a[0];
                let ascii = self.b.ins().icmp_imm_s(IntCC::UnsignedLessThan, cp, 0x80);
>>>>

# The runtime helpers (native/src/runtime/ops.rs)

@@ rs-alpha-alphabetic the helper's alpha is Rust's is_alphabetic (letter numbers and marks too)
## native/src/runtime/ops.rs
<<<<
    bool_value(tcl_alpha_char(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value")))
====
    bool_value(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value").is_alphabetic())
>>>>

@@ rs-alnum-is-alpha the helper's alnum is its alpha (non-ASCII digits are no alnum)
## native/src/runtime/ops.rs
<<<<
    bool_value(tcl_alnum_char(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value")))
====
    bool_value(tcl_alpha_char(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value")))
>>>>

@@ rs-alnum-numeric the helper's alnum is Rust's is_alphanumeric (every numeric category)
## native/src/runtime/ops.rs
<<<<
    bool_value(tcl_alnum_char(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value")))
====
    bool_value(char::from_u32(char_of(c)).expect("a UnicodeChar is a Unicode scalar value").is_alphanumeric())
>>>>

@@ rs-apply-op-swap the interpreter of NIR ops (apply_op) runs chartclalpha as alnum
## native/src/runtime/ops.rs
<<<<
        CharIsTclAlpha => rt_char_is_tcl_alpha(p, a[0]),
====
        CharIsTclAlpha => rt_char_is_tcl_alnum(p, a[0]),
>>>>
