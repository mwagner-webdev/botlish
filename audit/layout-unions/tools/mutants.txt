@@ union-read-cached a read of a live union's alternative is cached per view and offset in a function, as if native memory could not change between two reads
## native/lower.tcl
<<<<
proc native::lower::NativeField {fnVar e view field base} {
    upvar 1 $fnVar fn
    return [NativeRepr fn $e $view [lrange $field 2 end] [BasePlus fn $e $base [lindex $field 1]]]
}
====
proc native::lower::NativeField {fnVar e view field base} {
    upvar 1 $fnVar fn
    set key "$view/$base/[lindex $field 1]/[lrange $field 2 end]"
    if {[dict exists $fn readCache $key]} {
        return [dict get $fn readCache $key]
    }
    set r [NativeRepr fn $e $view [lrange $field 2 end] [BasePlus fn $e $base [lindex $field 1]]]
    dict set fn readCache $key $r
    return $r
}

@@ union-identity-process-wide whether a layout is a union is remembered process-wide by its identity, never reset or captured with a compilation's declarations (a same-named record of a later compilation is taken for the union)
## hir/layouts.tcl
<<<<
proc hir::layouts::IsUnion {id} {
    variable registry
    return [expr {[dict exists $registry $id] && [dict getdef [dict get $registry $id] union 0]}]
}
====
proc hir::layouts::IsUnion {id} {
    variable registry
    variable unionsEver
    if {![info exists unionsEver]} {
        set unionsEver [dict create]
    }
    if {[dict exists $registry $id] && [dict getdef [dict get $registry $id] union 0]} {
        dict set unionsEver $id 1
    }
    return [dict exists $unionsEver $id]
}
