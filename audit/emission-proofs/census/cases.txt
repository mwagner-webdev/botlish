# The modular census (MODULAR-EMISSION-PROOFS-MILESTONE-1.md, section 14):
# programs written to expose unsound or caller-dependent reasoning, each
# checked by the modular checker (tools/modcheck.tcl) and by the
# exploratory one (audit/emitter/tools/emitcheck.tcl), every rejected claim
# searched for a concrete counterexample (tools/concrete.tcl), every proven
# one run on the same candidates (tools/census.tcl).
#
#   @@ NAME DESCRIPTION
#   expect FN VERDICT           the modular verdict: proven, disproven,
#                               not-proved
#   original FN VERDICT         the exploratory checker's: proved,
#                               rejected, unused (with the case's own
#                               callers -- which pass valid values only:
#                               its proofs that are false for the
#                               declaration are the census's subject)
#   --- FILE                    a file of the case (main.bot is the entry;
#                               it is added beside the JSON workspace)
#   ...
#   @@end

@@ caught-before-yield a helper whose guard fails before it yields; the caller catches it and writes a digit
expect safe proven
original safe proved
--- main.bot
import std::dfa
import json_grammar

error NotDigit:
    at: int

#: emits json_grammar::digit_symbol
fn guarded(c: int) -> unit errors NotDigit:
    json_grammar::digit_symbol(c):
        on std::dfa::UnrecognizedInput:
            fail NotDigit {at: 0}
        on std::dfa::UnexpectedEnd:
            fail NotDigit {at: 0}
    yield c
    unit

#: emits json_grammar::digit_symbol
fn safe(c: int) -> unit:
    guarded(c):
        on NotDigit:
            yield 48
    unit

fn src() -> int:
    safe(65)
    -1

coroutine {step, first} = src()
first
@@end
