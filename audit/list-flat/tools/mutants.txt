# Mutants of LIST-FLAT-MILESTONE-1.md (audit/list-flat/tools/mutate.tcl):
# each a deliberate bug in std::list::flat -- inner Lists or their elements
# reversed, an empty inner List kept or ending the result, two levels
# removed, an element dropped or duplicated, the total length miscounted,
# the result built by copying an expanding prefix (quadratic), the element
# kind check gone -- in the reference implementation (core/lists.tcl) and
# the native runtime (native/src/runtime/ops.rs); and in the static side:
# the result's type, its length refinement, the element-kind rule, exact
# values, callable contracts, module bindings, the ownership role and the
# native lowering.
#
#   @@ NAME DESCRIPTION
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# A mutant may have several edits. Lines starting with # outside an edit
# are comments. Update a mutant's text when the code it mutates changes,
# keeping it a mutant of the same rule. A mutant that edits native/src/ is
# a native mutant: the harness rebuilds a private copy of the native
# backend for it.

@@ rs-quadratic the native runtime builds the result by copying the expanding prefix once per inner List (same output, O(n*m) copies)
## native/src/runtime/ops.rs
<<<<
    let mut items: Vec<Value> = Vec::with_capacity(total);
    for &inner in outer {
        items.extend_from_slice(list_of(inner).items());
    }
====
    let mut items: Vec<Value> = Vec::new();
    let mut copied: usize = 0;
    for &inner in outer {
        let mut next: Vec<Value> = Vec::with_capacity(items.len() + list_of(inner).items().len());
        next.extend_from_slice(&items);
        next.extend_from_slice(list_of(inner).items());
        copied += next.len();
        items = next;
    }
    let total = copied;
>>>>
