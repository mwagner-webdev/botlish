# the typed constructor (scratch control: nothing in the canonical corpus).

        grown = mutarray::create(length * 2 + 1, value)
        mutable_array_copy(grown, 0, storage, 0, length)
        mutable_array_set(grown, length, value)

### shared-typed-builder-loop
fn tb_fill(storage, xs, i):
    if i == list_length(xs):
        storage
    else:
        mutable_array_set(storage, i, list_get(xs, i))
        tb_fill(storage, xs, i + 1)
a = tb_fill(mutarray::create(2, "z"), ["p", "q"], 0)
b = tb_fill(mutarray::create(2, 0), [1, 2], 0)
[mutable_array_freeze(a, 2), mutable_array_freeze(b, 2)]
