# Mutants of LOOP-CLAUSE-BINDING-TYPE-CORRECTNESS-MILESTONE-1.md
# (audit/loop-binding-types/tools/mutate.tcl): deliberate breaks of the rule
# that a loop clause binding -- a listloop's element, a countloop's counter,
# a lockstep clause -- records the type the checker gives its references
# (hir/types.tcl's RecordClause), that the program's record is its generic
# walk's and an instance's its snapshot's (hir/semantic.tcl), and that the
# compiler service, HIR interchange, HIR text and the front-end cache carry
# it. Each must be killed by one of the tests its `!!` line names (a pattern
# of test names), run from the test files its `%%` line names (default
# tests/loop-clause-binding-types.test), never by an unrelated failure.
#
#   @@ NAME DESCRIPTION
#   !! TEST-PATTERN ...
#   %% TEST-FILE ...
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>

# Semantic instances (hir/types.tcl's Walk, hir/semantic.tcl's Snapshot)

@@ lcb-first-instance-global the first semantic instance's loop variable types are stored as the program's records: a generic function's element reads the first call's element type, and which one depends on the order of the calls
!! lcb-generic-record lcb-service-generic lsp-lang-loop-variable-*
%% tests/loop-clause-binding-types.test tests/lsp.test
## hir/types.tcl
<<<<
    set hir $typed
    dict set hir semantic $snapshot
}
====
    set hir $typed
    dict set hir semantic $snapshot
    set taken {}
    dict for {id inst} [dict getdef $snapshot instances {}] {
        if {![dict exists $inst snapshot bindings]} continue
        dict for {b t} [dict get $inst snapshot bindings] {
            if {[dict exists $taken $b] || [dict get $hir bindings $b kind] ne "param"} continue
            dict set taken $b 1
            dict set hir bindings $b type [intern hir $t]
        }
    }
}
>>>>

====
proc service::BindingType {hir b} {
    set binding [dict get $hir bindings $b]
    set loop [dict get $hir scopes [dict get $binding scope] owner]
    if {$loop ne "" && $b in [hir::loopClauseBindings [dict get $hir exprs $loop]]} {
        return any
    }
