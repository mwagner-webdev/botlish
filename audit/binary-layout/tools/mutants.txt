# Mutants of fixed binary layouts (audit/binary-layout/tools/mutate.tcl;
# BINARY-LAYOUT.md): each a deliberate bug in the declaration checks
# (hir/layouts.tcl), the type lattice (hir/types.tcl), resolution
# (hir/resolve.tcl), the HIR pipeline (hir/hir.tcl), scalar replacement
# (hir/escape.tcl), the interpreter's reference implementation
# (core/binary.tcl, whose Letter the Tcl compiler shares), the Tcl compiler
# (compiler/compiler.tcl), native lowering (native/lower.tcl) or the native
# runtime (native/src/runtime/ops.rs). The brief's mutants come first, one per
# layer where a layer implements the rule separately, then further ones.
#
#   @@ NAME DESCRIPTION
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# A mutant may have several edits. A mutant editing native/src/ or the
# native crate's manifest is a native mutant: mutate.tcl rebuilds the native
# backend for it. Lines starting with # outside an edit are comments.
#
# Brief mutants with no separate non-equivalent mutant in this
# representation:
#
#   * "emit truncates a value too wide for its field" on the Tcl backends:
#     a field's value is in its domain by its type (construction is checked
#     like any struct's, statically), so no checked program reaches
#     core/binary.tcl's RANGE check -- removing it is an equivalent mutant
#     there. The native store's own check is exercised directly by the Rust
#     runtime tests (store-truncates-rust).

# --- ignore field offset ------------------------------------------------

@@ offset-ignored-interp the interpreter reads every scalar field at its layout's base, ignoring its offset
## core/binary.tcl
<<<<
        lappend values [DecodeRepr [lrange $field 2 end] $bytes [expr {$base + [lindex $field 1]}] $name $structId]
====
        lappend values [DecodeRepr [lrange $field 2 end] $bytes $base $name $structId]
>>>>

@@ offset-ignored-native native parse loads every scalar field at its layout's base, ignoring its offset
## native/lower.tcl
<<<<
        lappend registers [ReprRead fn $e $mode $source [lrange $field 2 end] [BasePlus fn $e $base [lindex $field 1]]]
====
        lappend registers [ReprRead fn $e $mode $source [lrange $field 2 end] $base]
>>>>

@@ nested-offset-ignored-native native emit writes a nested layout's fields at its parent's base, ignoring the nested field's offset
## native/lower.tcl
<<<<
            lappend leaves {*}[LayoutLeaves [lindex $field 3] [expr {$base + $offset}]]
====
            lappend leaves {*}[LayoutLeaves [lindex $field 3] $base]
>>>>

@@ emit-sized-by-fields-native native emit allocates up to the end of the last field instead of the layout's size
## native/lower.tcl
<<<<

# --- write fields sequentially despite gaps ----------------------------

@@ sequential-emit-interp the interpreter's emit writes fields one after another from the base, ignoring gaps
## core/binary.tcl
<<<<
    foreach field $fields {
        set name [lindex $field 0]
        set bytes [EncodeRepr std::binary::emit [lrange $field 2 end] [core::value::structGet $value $name] \
            $bytes [expr {$base + [lindex $field 1]}] "field \"$name\" of $structId"]
    }
====
    set cursor $base
    foreach field $fields {
        set name [lindex $field 0]
        set bytes [EncodeRepr std::binary::emit [lrange $field 2 end] [core::value::structGet $value $name] \
            $bytes $cursor "field \"$name\" of $structId"]
        switch -- [lindex $field 2] {
            layout { set cursor [expr {$cursor + [lindex $field 3 1]}] }
            array { set cursor [expr {$cursor + [lindex $field 3] * [lindex $field 4]}] }
            default { set cursor [expr {$cursor + [lindex $field 3]}] }
        }
    }
>>>>

# --- leave padding non-deterministic -------------------------------------

@@ padding-interp the interpreter's emit starts from bytes that vary by time, not zeros
## core/binary.tcl
<<<<
[binary format x$size]
====
[string repeat [format %c [expr {1 + [clock microseconds] % 255}]] $size]
>>>>

@@ padding-compile the Tcl compiler's emit fills the bytes no field covers with 0x55
## compiler/compiler.tcl
<<<<
            append format @$size
            set t [NewTemp]
====
            append format @$size
            set format "a$size@0$format"
            set words [linsert $words 0 [string repeat U $size]]
            set t [NewTemp]
>>>>

@@ padding-rust byteszeroed fills the fresh storage with a byte that varies by time, not zeros
## native/src/runtime/ops.rs
<<<<
    vm(p).new_bytes_with(count, |init| init.push_zeros(count))
====
    let noise = (std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH).map(|d| d.subsec_nanos()).unwrap_or(7) as u8) | 1;
    vm(p).new_bytes_with(count, |init| {
        for _ in 0..count {
            init.push(noise)
        }
    })
>>>>

# --- treat LayoutStruct[A] as LayoutStruct[B] ----------------------------

@@ structural-layout-compat LayoutStruct[A] is a subtype of LayoutStruct[B] when the two layouts have the same field names and types
## hir/types.tcl
<<<<
proc hir::types::subtype {a b} {
    if {$a eq "never" || $b eq "any" || $a eq $b} {
        return 1
    }
====
proc hir::types::subtype {a b} {
    if {$a eq "never" || $b eq "any" || $a eq $b} {
        return 1
    }
    if {[hir::layouts::OfType $a] ne "" && [hir::layouts::OfType $b] ne ""
            && [hir::layouts::StructFieldTypes [hir::layouts::OfType $a]] eq [hir::layouts::StructFieldTypes [hir::layouts::OfType $b]]} {
        return 1
    }
>>>>
