# Mutants of the native link plan (audit/native-link-plan/tools/mutate.tcl;
# NATIVE-LINK-PLAN-MILESTONE-1.md): each a deliberate bug in the declaration
# grammar's lowering (surface/lower.tcl), the loader's placement of a
# declaration (surface/modules.tcl), the plan's merge (hir/foreign.tcl), HIR
# interchange (hir/interchange.tcl), the front-end cache's key
# (surface/cache.tcl), the NIR header (native/lower.tcl), the driver's
# options (native/native.tcl) or the driver's resolution, checks and
# arguments (native/src/codegen/aot.rs). The milestone's table first, then
# further ones.
#
#   @@ NAME DESCRIPTION
#   %% TARGET ...          (optional: the hosts it runs on)
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# Lines starting with # outside an edit are comments.

# --- the milestone's table ---------------------------------------------------

@@ static-dropped-in-lowering a declaration's `static` is lost between the AST and HIR: the library is the linker's choice again
## surface/lower.tcl
<<<<
        linkage [dict get $node linkage] path [dict get $node path] base "" namespace $namespace \
====
        linkage default path [dict get $node path] base "" namespace $namespace \
>>>>

<<<<
proc hir::interchange::WithoutRepresentations {hir} {
====
proc hir::interchange::WithoutRepresentations {hir} {
    if {[dict exists $hir links]} {
        dict set hir links [lmap entry [dict get $hir links] {
            if {[dict exists $entry linkage]} {
                dict set entry linkage default
                dict set entry path ""
            }
            set entry
        }]
    }
