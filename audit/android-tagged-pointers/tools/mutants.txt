# Mutants of the target's C address encoding and of byte exposures made by
# the C call (audit/android-tagged-pointers/tools/mutate.tcl;
# RN-ANDROID-TAGGED-POINTER-CORRECTNESS-MILESTONE-1.md, "Mutation testing"):
# the encoding (native/src/runtime/value.rs), its one statement per target
# and the target's fingerprint (codegen/target.rs, runtime/aotimage.rs,
# codegen/mod.rs's program record, main.rs's runtime-abi), generated code's
# encode and decode (codegen/clif.rs), the exposure the C call makes
# (codegen/clif.rs's exposure_address, native/lower.tcl's BorrowCall), and
# the compiler's comparisons with a target's runtime (native/support.tcl,
# native/library.tcl).
#
#   @@ NAME DESCRIPTION
#   !! TEST ...        the tests that must kill it (at-* android-target.test,
#                      cf-* c-ffi.test, cb-* c-callbacks.test; rust:FILTER
#                      the native crate's cargo tests matching FILTER)
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# A mutant is killed only by a detector its !! line names failing; a mutant
# that does not build is NOT-BUILT, never killed. The at-* detectors need
# ANDROID_NDK_ROOT, qemu-aarch64 and BOTLISH_ANDROID_ROOTS (and, for
# at-tagged-sdl3, Android's libSDL3.so: tests/cffi/sdl3.tcl android).

# Full-width pointers: the encoding

@@ decode-masks-tag the tagged encoding's decode clears the top byte (a tag "normalized" away)
!! rust:address_word_tests at-tagged-pointers
## native/src/runtime/value.rs
<<<<

====
        ((v as i64) >> 4).rotate_right(self.rotation()) & 0x00ff_ffff_ffff_ffff

!! rust:address_word_tests at-tagged-pointers

        (((address & 0x00ff_ffff_ffff_ffff).rotate_left(self.rotation()) as u64) << 4) | CPTR_TAG

# Generated code: the inline encode and decode

@@ codegen-decode-unrotated generated code decodes a C address word with the shift alone (Android's pointers handed to C rotated)
!! rust:android_code_rotates_by_the_top_byte at-tagged-pointers
## native/src/codegen/clif.rs
<<<<
        let shifted = self.b.ins().sshr_imm_s(v, 4);
        match self.address_word.rotation() {
            0 => shifted,
            r => self.b.ins().rotr_imm_u(shifted, r as i64),
        }
====
        self.b.ins().sshr_imm_s(v, 4)
>>>>

@@ codegen-encode-unrotated generated code encodes a C pointer result without the rotation (a tagged result refused, or a wrong word)
!! rust:android_code_rotates_by_the_top_byte at-tagged-heap-pointer
## native/src/codegen/clif.rs
<<<<
    fn cptr_rotated(&mut self, address: ir::Value) -> ir::Value {
        match self.address_word.rotation() {
            0 => address,
            r => self.b.ins().rotl_imm_u(address, r as i64),
        }
    }
====
    fn cptr_rotated(&mut self, address: ir::Value) -> ir::Value {
        address
    }
>>>>
