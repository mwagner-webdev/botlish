# The rule itself

@@ npm-no-value-parameters no binding is a value parameter: every mutation of an ordinary parameter compiles again and silently changes the callee's copy (the SimCity defect)
!! npm-direct npm-every-mutating-operation npm-simcity-field-helper npm-simcity-copy-destination npm-simcity-byte-module
## hir/mutvec.tcl
<<<<
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
    return [expr {$index >= 0 && $index ni [dict getdef $hir exprs $f places {}]}]
====
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
    return 0
>>>>

# What is not a value parameter

@@ npm-place-parameters-too a place parameter counts as a value parameter: the caller's own storage can no longer be updated through one
!! npm-mixed-parameters npm-imprinted npm-cross-module-accepted npm-simcity-field-helper
## hir/mutvec.tcl
<<<<
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
    return [expr {$index >= 0 && $index ni [dict getdef $hir exprs $f places {}]}]
====
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
    return [expr {$index >= 0}]
>>>>

@@ npm-construct-bindings-too every parameter binding counts, not only a function's: a loop variable and a handler's payload binding (parameters of their construct's scope) are rejected
!! npm-construct-bindings
## hir/mutvec.tcl
<<<<
    if {$f eq "" || ![dict exists $hir exprs $f] || [dict get $hir exprs $f kind] ne "block"} {
        return 0
    }
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
====
    if {$f eq "" || ![dict exists $hir exprs $f] || [dict get $hir exprs $f kind] ne "block"} {
        return 1
    }
    set index [lsearch -exact [dict get $hir exprs $f params] $b]
>>>>

@@ npm-context-parameters-too a context parameter counts as a value parameter: the installed context's members can no longer be mutated
!! npm-context-parameter npm-reads-and-other-effects
## hir/mutvec.tcl
<<<<

        return 0

====
proc hir::mutvec::IsValueParam {hir b} {
    if {$b eq "" || ![dict exists $hir bindings $b]} {
        return 0
    }
    if {[IsContextParam $hir $b]} {
        return 1
    }
